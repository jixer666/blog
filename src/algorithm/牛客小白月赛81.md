## 小辰的圣剑

这题卡了半天，后面发现数据范围很小，两个循环暴力拿下

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
//int a, b, x, y, res;
//int n[N];

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	ll n, m, u, res = 0;
	cin >> n >> m >> u;
	vector<int> v(n), n1(n);
	for(int i = 0; i < n; i++) cin >> v[i];
	for(int i = 0; i < n; i++) cin >> n1[i];
	for(int i = 0; i < n; i++){
		ll cot = 0, lj = 0, ry = 0;
		for(int j = i; j < n; j++){
			if(v[j] + lj <= m && n1[j] + ry <= u){
				cot ++;
				lj += v[j];
				ry += n1[j];
			} else {
				break;
			}
		}
		res = max(res, cot);
	}
	cout << res << endl;
	
	return 0;
}
```

## 陶陶学算术

这题参考题解，数据范围很大，除的话会超出范围，~~我一直卡在这不知道范围问题~~

我们需要对这个公式进行改变

res1 = x1x2x3/y1y2y3 = A / B

res2 = i1i2i3/j1j2j3 = C / D

=> A * D = B * C ，将除法转为乘法并用long long存放

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, res;
int n[N];

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a;
	ll cot1 = 1, cot2 = 1;
	while(a --){
		cin >> x >> y;
		if(x == 1) cot1 = cot1 * y % M;
		else cot2 = cot2 * y % M; 
	}
	cin >> b;
	while(b --){
		cin >> x >> y;
		if(x == 1) cot2 = cot2 * y % M;
		else cot1 = cot1 * y % M; 
	}
	cout << (cot1 == cot2 ? "YES" : "NO") << endl;
	return 0;
}
```

## 小辰的借钱计划

这题用到数学期望，不会

期望 = 1 / 符号条件的 * 符合条件的数

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, res;
int n[N];

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a;
	while(a --){
		cin >> x >> y;
		int idx = 0;
		for(int i = 1; i <= x - y; i++){
			if(y % i == 0 || i % y == 0){
				idx ++;
			}
		}
		double w = 0.0, ra = 1.0 / idx;
		for(int i = 1; i <= x - y; i++){
			if(y % i == 0 || i % y == 0){
				w += i * ra;
			}	
		}
		cout << (w > y ? "YES":"NO") << endl;	
	}
	return 0;
}
```
