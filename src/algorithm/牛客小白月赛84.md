## k级序列

设当前下标为i，分两种情况

- 当前数比前一个数大的时候，就取 min(v[i - 1], v[i] - k);
- 当前数比前一个数小的时候
  - 若 v[i] + k 任然小于 v[i - 1] 就是不满足条件的输出No
  - 否则，就取 min(v[i] + k, v[i - 1]);

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, z;


int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a;
	while(a--){
		cin >> x >> y;
		vector<int> v(x);
		for(int i = 0; i < x; i++){
			cin >> v[i];
		}
		if(x == 1){
			cout << "Yes" << endl;
			continue;
		}
		int flag = 1;
		v[0] -= y; 
		for(int i = 1; i < x; i++){
			if(v[i] >= v[i - 1]){
				v[i] = max(v[i] - y, v[i - 1]);
			} else {
				if(v[i] + y < v[i - 1]){
					flag = 0;
					break;	
				}
				v[i] = min(v[i] + y, v[i - 1]);
			}
		}
		cout << (flag ? "Yes" : "No") << endl;

	}
	
	return 0;
}
```

## Reverse

先遍历找出现在有多少连续的1段，设置为 res。两端分分开判断，其次判断中间区间

设当左右下标为x，y，分情况讨论，这里只讨论 v[x] != v[y]，因为相等的情况一致讨论没意义

- 若 x == 1，若判断 v[x] == 1并且 v[y + 1] == v[x]，就需要 res--；若 v[x] == 0 并且 v[y + 1] != v[x]，则 res ++;
- 若 y == a，判断同上
- 若 v[x - 1] == v[x] 并且 v[y] == v[y + 1]，则 res --;  若v[x - 1] != v[x] 并且 v[y] != v[y + 1]，则 res ++;

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, z;

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	int res = 0, flag = 0;
	cin >> a >> b;
	vector<char> v(a);
	for(int i = 1; i <= a; i++){
		cin >> v[i];
		if(v[i] == '1' && !flag){
			res ++, flag = 1;
		} else if(v[i ] == '0') flag = 0;
	}
	while(b --){
		cin >> x >> y;
		int cot = res;
		if(v[x] != v[y]){
			if(x == 1){
				if(y != a && v[x] == '0' && v[y + 1] != v[x]) cot ++;
				else if(y != a && v[x] == '1' && v[y + 1] == v[x]) cot --;
			} else if(y == a){
				if(x != 1 && v[y] == '0' && v[x - 1] != v[y]) cot ++;	
				else if(x != 1 && v[y] == '1' && v[x - 1] == v[y]) cot --;			
			} else {
				if(v[x - 1] == v[x] && v[y] == v[y + 1]){
					cot ++; 
				} else if(v[x - 1] != v[x] && v[y] != v[y + 1]){
					cot --;
				}
			}
		}
		cout << cot << endl;
	}
	
	return 0;
}
```

## Dog vs Cat

这题也是需要分开讨论，难就难在需要讨论个数为2的特殊情况

- 当个数为2的时候
  - 若两个数相差1，且最小的不是0，输出 Cat
  - 否则输出Dog
- 其他情况：算出所有数与1的差之和，设为 res；计算数组个数的一半，若是奇数需要 / 2 + 1，设为 cc；计算 res + cc 是否是奇数，若是奇数输出 Dog，否则输出 Cat

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, z;

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	int a;
	cin >> a;
	while(a --){
		cin >> b;
		vector<int> v(b);
		ll res = 0;
		int cot = 0;
		for(int i = 0; i < b; i++){
			cin >> v[i];
			if(v[i] != 0){
				res += v[i] - 1;
			} else {
				cot ++;
			}
		}
		if(b == 2){
            sort(v.begin(),v.end());
			if(v[0] && v[0] + 1 == v[1]) cout << "Cat" << endl;
			else cout << "Dog" << endl;
			continue;
		}
        
        int ban;
        if(b % 2 == 0) ban = b / 2;
        else ban = b / 2 + 1;
		if(cot >= ban){
			cout << "Dog" << endl;
		} else {
			int cc = ban - cot;
			if((res + cc) % 2 == 0) cout << "Cat" << endl;
			else cout << "Dog" << endl; 
		}
	
	}
	
	return 0;
}
```
