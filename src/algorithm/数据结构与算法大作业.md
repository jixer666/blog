// 找前驱结点，结果放到path数组中 
void dijkstra(string s1, string s2, int path[], int &min_dis){
	int x = get_int(s1);
	int y = get_int(s2);
	int dis[N], s[N], final[N];
	for(int i = 0; i < vertexNum; i++){
		final[i] = 0;
		dis[i] = point[x][i];
		if(dis[i] != MAX) path[i] = x; 
        else path[i] = -1;
	}
	s[0] = x;
	final[x] = 1;
	int flag = 1;
	while(flag < vertexNum){
		int k = 0, minn = MAX;
		for(int i = 0;i < vertexNum; i++){
			if(i == x) continue;
			if(!final[i] && dis[i] < minn){
				k = i;
				minn = dis[i];
			}
		}
		s[flag++] = k;
		final[k] = 1;
		for(int i = 0; i < vertexNum; i++){
			if(!final[i] && dis[i] > dis[k] + point[k][i]){
				dis[i] = dis[k] + point[k][i];
				path[i] = k;
			}
		}
	}
	min_dis = dis[y];  // 存入最短长度 
}

// 找最短路径
void get_path(int x, int y, int path[], int min_path[], int& i){
	if(x == y) return;
	min_path[i++] = path[y];  // 此时的path[y]就存放的该时刻的最短路径 
	get_path(x, path[y], path, min_path, i);
}

// 格式化输出最短路径 
void query(string s1, string s2, int path[], int min_dis){
	int min_path[N];  // min_path数组存放最短路径的最终结果 
	int x = get_int(s1);
	int y = get_int(s2);
	int i = 0;
	min_path[i++] = y; // 从终点开始往根据数组值前找起点，存入min_path中
	get_path(x, y, path, min_path, i); 
	cout << "最短路径：";
	for(int j = i - 1; j >= 0; j--){  // 逆序输出min_path即可 
		int w = min_path[j];
		if(j == 0) cout << get_string(w) << endl;		
		else cout << get_string(min_path[j]) << "->";
	} 
	cout << "最短长度：" << min_dis << endl;
}

    char data;         								// 结点值
    int weight;        								// 权重
    int parent;        								// 父结点
    int lchild;      								// 左孩子
    int rchild;      								// 右孩子
//    string er;										//二进制编码 
}HTNode;


typedef struct {     			// 定义Huffman编码结构

bool cmp(SS a,SS b){
	return a.n<b.n;                   				//按照字符数量排序
}


void CreatHT(HTNode *HT, int n){
	int i,k,l,r;
	int m1, m2;
    for(i = 1;i<2*n; i++) 
        HT[i].parent = HT[i].lchild = HT[i].rchild = 0;						// 初始化
    for(i = n+1; i<2*n; i++) {    											// 构造Huffman树
        m1 = m2 = 0x7fff;													// m1取最小权重，m2取次小权重
        int lnode = 0,rnode = 0;     										// lnode, rnode分别取两个最小权重的结点位置
        for( k=1; k<i; k++)
		{	if(HT[k].parent==0)
			{
				if(HT[k].weight<m1) {
	                m2 = m1;
					rnode=lnode;
					m1 = HT[k].weight;
					lnode=k;
				}
				else if(HT[k].weight<m2) {
					m2 = HT[k].weight;
					rnode = k;
				}
            }
		}
        HT[lnode].parent = i;
		HT[rnode].parent = i;
		HT[i].weight =  HT[lnode].weight+HT[rnode].weight;
		HT[i].lchild = lnode;
		HT[i].rchild = rnode;
    }
}

void CreatHCode(HTNode *HT, HCode *hcd, int n ){
	int i, f, c;
	HCode hc;
    for(i=1; i<=n; i++) 
    {
		hc.start = n;
        c = i;
        f = HT[i].parent;
        while( f!=0 ) {
            if(HT[f].lchild==c)   										// c是f的左孩子，编码取'0'，否则取'1'
                hc.cd[--hc.start]='0';
            else
                hc.cd[--hc.start]='1';
            c = f;														// 向根结点方向搜索
			f = HT[f].parent;
        }
		hcd[i]= hc;
    }
}

    int parent;        		// 父结点
    int lchild;      			// 左孩子
    int rchild;      			// 右孩子
}HTNode;
typedef struct {     			// 定义Huffman编码结构
    char cd[MAX]; 
    int start; 
}HCode;

void CreatHT(HTNode *HT, int n)
{
	int i,k,l,r;
	int m1, m2;
    for(i=1;i<2*n; i++) 
        HT[i].parent = HT[i].lchild = HT[i].rchild = 0;	// 初始化
    for(i=n+1; i<2*n; i++) {    		// 构造Huffman树
        m1 =m2 = 0x7fff;			// m1取最小权重，m2取次小权重
        int lnode  = 0 , rnode = 0;     		// lnode, rnode分别取两个最小权重的结点位置
        for( k=1; k<i; k++)
		{	if(HT[k].parent==0)
			{
				if(HT[k].weight<m1) {
	                int m2 = m1;  rnode=lnode;
					m1 =  HT[k].weight;  lnode=k;
				}
				else if(HT[k].weight<m2) {
					m2 =HT[k].weight;
					rnode =  k ;
				}
            }
		}
        HT[lnode].parent = i;
		HT[rnode].parent = i;
		HT[i].weight =  HT[lnode].weight+HT[rnode].weight;
		HT[i].lchild = lnode;
		HT[i].rchild = rnode;
    }
}

void CreatHCode(HTNode *HT, HCode *hcd, int n ){
	int i, f, c;
	HCode hc;
    for(i=1; i<=n; i++) 
    {
		hc.start = n;
        c = i;
        f = HT[i].parent;
        while( f!=0 ) {
            if(HT[f].lchild==c)   										// c是f的左孩子，编码取'0'，否则取'1'
                hc.cd[--hc.start]='0';
            else
                hc.cd[--hc.start]='1';
            c = f;														// 向根结点方向搜索
			f = HT[f].parent;
        }
		hcd[i]= hc;
    }
}

void PrintHCode(HTNode *HT, HCode *hcd, int n)
{
	int i,k;
	for(i=1; i<=n; i++) 
	{
		printf("  %c：",HT[i].data);
		for(k= hcd[i].start; k<n; k++)
			printf("%c", hcd[i].cd[k]);
		printf("\n");
	}
}
