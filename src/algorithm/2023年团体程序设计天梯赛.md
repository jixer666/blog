```c++
		int m, n;
		cin >> m >> n >> s1 >> s2;
		string sq = str.substr(m - 1, n - m + 1);
		string res_ = str.substr(0, m - 1) + str.substr(n, len);
		string all = s1 + s2;
		int f = res_.find(s1), s = res_.find(s2), fs = res_.find(all);

		}
	}
```

## **L1-095 分寝室**

这题没考虑到 `不允许单人住一间寝室` 这一条件和浮点数问题(分母为0了)。

加个边界判断`i <= x - 1` 解决了浮点数问题，` a != i && b != x - i` 解决单人住一间的问题

```c++
#include<bits/stdc++.h>
#define line printf("------\n")
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, res; 

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	res = 100010; 
	int x, r = 0, l = 0;
	cin >> a >> b >> x;
	for(int i = 1; i <= x - 1; i++){
		if(a % i == 0 && b % (x - i) == 0 && a != i && b != x - i){
			int cot = fabs(a / i - b / (x - i)); 
			if(cot < res){
				res = cot;
				r = i, l = x - i;
			}
		}
	}
	if(l == 0 && r == 0) cout << "No Solution" << endl;
	else cout << r << " " << l << endl;
	
	
	return 0;
}	
```

## **L2-045 堆宝塔**

```c++

	int num = 0, maxx = 0;
	cin >> a;
	stack<int> s, s2;
	vector<int> v(a);
	for(int i = 0; i < a; i++) cin >> v[i];
	for(int i = 0; i < a; i++){
		if(!s.empty()){
			if(s.top() > v[i]) s.push(v[i]);
			else {
				if(s2.size()){
					if(s2.top() < v[i]){
						s2.push(v[i]);
					} else {
						int cot = 0;
						while(s.size()){
							s.pop();
							cot ++;
						}
						maxx = max(maxx, cot);
						num ++;
						while(s2.size() && s2.top() > v[i]){
							s.push(s2.top());
							s2.pop();	
						}
						s.push(v[i]);
					}
				} else {
					s2.push(v[i]);
				}
			}
		} else {
			s.push(v[i]);
		}
	}
	int cot = s.size();
	if(cot){
		num ++;
		maxx = max(maxx, cot);
	}
	cot = s2.size();
	if(cot){
		num ++;
		maxx = max(maxx, cot); 
	}
	cout << num << " " << maxx << endl;
```

## **L2-046 天梯赛的赛场安排**

理解题意还是能混点分的

```c++
#include<bits/stdc++.h>
#define line printf("------\n")
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b; 


int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a >> b;
	int res = 0;
	priority_queue<int> q;
	for(int i = 0; i < a; i++){
		string str;
		int num;
		cin >> str >> num;
		q.push(num % b);
		res += num / b;
		cout << str << " " << ceil(num * 1.0 / b) << endl;
	}
	int cot = 0, flag = 1;
	while(q.size()){
		int num = q.top();
		q.pop();
		if(cot + num <= b){
			cot += num;
			flag = 1;
		} else {
			res ++;
			cot = num;
			flag = 0;
		}
	}
	if(!flag){
		res ++;
	}
	cout << res << endl;
 
	return 0;
}
```

## **L2-048 寻宝图**

dfs模板题

```c++
#include<bits/stdc++.h>
#define line printf("------\n")
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, flag; 
char n[1000][1000];
int st[1000][1000];

void dfs(int x, int y){
	if(n[x][y] > '1'){
		flag = 1;
	}
	
	for(int i = 0; i < 4; i++){
		int xx = ix[i] + x, yy = iy[i] + y;
		if(xx >= 0 && xx < a && yy >= 0 && yy < b && !st[xx][yy] && n[xx][yy] >= '1'){
			st[xx][yy] = 1;
			dfs(xx, yy);
		}
	}
}


int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a >> b;
	int res = 0, cot = 0;
	for(int i = 0; i < a; i++){
		for(int j = 0; j < b; j++){
			cin >> n[i][j];
		}
	}
	for(int i = 0; i < a; i++){
		for(int j = 0; j < b; j++){
			if(n[i][j] >= '1' && !st[i][j]){
				flag = 0;
				st[i][j] = 1;
				dfs(i, j);
				res ++;
				if(flag){
					cot ++;
				}
			} 
		}
	}
	cout << res << " " << cot << endl;

 
	return 0;
}
```
