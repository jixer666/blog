                                                                                                            const int INF=2*int(1e9)+10;
                                                                                                            #define LL long long

                                                                                                                return a>b;
                                                                                                            }
                                                                                                            int main(){
                                                                                                                int num[6]={1,2,4,7,15,34}; 
                                                                                                                sort(num,num+6);                           //按从小到大排序 
                                                                                                                int pos1=lower_bound(num,num+6,7)-num;    //返回数组中第一个大于或等于被查数的值

                                                                                                                int pos3=lower_bound(num,num+6,7,greater<int>())-num;  //返回数组中第一个小于或等于被查数的值
