## 采矿时间到！

wa了一次，这题坑点在于会有一种这样的情况

```
#################*##
#########*##########
....................
#####*######**######
#*##########*#######
```

最外层和最内层都有矿石，此时就只需要消耗2体力就能够挖到两个矿石

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e5 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, z, res;
 char v[1010][1010];

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	int idx = 0;
	cin >> a >> b;
	for(int i = 0; i < 5; i++){
		for(int j = 0; j < a; j++){
			cin >> v[i][j];
			if(v[i][j] == '*') idx ++;
		}
	}
	int cot = 0, res = 0;
	for(int i = 0; i < a; i++){
		if(v[1][i] == '*') cot ++;
		if(v[3][i] == '*') cot ++; 
	} 
	if(cot > b){
		cout << b << endl;
		return 0; 
	}
	int cc = (b - cot) / 2;
	cout << min(cc, idx - cot) + cot << endl;
	return 0;
}
```

## 耕种时间到！

这题是个模拟题，很容易想到模拟的过程

我在赛时把分解的过程直接化了，简单的想成了小麦种子的数量为`收割次数总次数 * 2`，正解是`pow(2, 收割次数总次数 )`

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e5 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, z;

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	int idx = 0;
	cin >> a;
	vector<int> v(a);
	unordered_map<int, ll> m; 
	for(int i = 0; i < a; i++){
		cin >> v[i];
	}
	cin >> b;
	for(int i = 0; i < a; i++){
		ll idx = 0, cnt = 1;
		while(v[i] > b){
			v[i] = ceil(v[i] * 1.0 / 3);
			idx ++;
			cnt *= 2;
		}
		if(v[i] == b){
			m[idx] += cnt;
		}
	}
	ll res = 0;
	for(auto i: m){
		res = max(res, i.second);
	}
	cout << res << endl;
	
	return 0;
}
```

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e5 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, z, res;

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a;
	vector<int> v(a + 1);
	ll all = 0, all2 = 0, all3 = 0, res = 0x3f3f3f3f3f3f3f3f;
	for(ll i = 1; i <= a; i++){
		cin >> v[i];
		all += v[i];
		all2 += i * v[i];
		all3 += i * i * v[i];
	}
	for(ll i = 1; i <= a; i++){
		res = min(res, i * i * all + all3 - 2 * i * all2);
	}

	cout << res << endl; 
	
	
	return 0;
}
```

## 来硬的

背包问题

首先从数据范围上很容易看出是DP问题，需要双循环更新

~~其次我知道它是DP，但我设计不出来状态转移方程，淦~~

设计状态：`dp[i][j][0/1]` 表示前i个煤炭，获得j个矿石的最短时间，0/1表示是否使用过魔法

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, z;

struct p{
	int x, y;
}n[N];

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a >> b;
	vector<vector<vector<ll>>> dp(a + 1, vector<vector<ll>>(b + 1, vector<ll>(2, 0x3f3f3f3f3f3f3f3f)));
	dp[0][0][0] = 0;
	for(int i = 1; i <= a; i++){
		cin >> n[i].x >> n[i].y;
	}
	for(int i = 1;  i <= a; i++){
		for(int j = 0; j <= b; j++){
			dp[i][j][0] = min(dp[i - 1][j][0], dp[i - 1][max(j - n[i].x, 0)][0] + n[i].y);
			dp[i][j][1] = min({dp[i - 1][j][1], dp[i - 1][max(j - n[i].x * 2, 0)][0] + n[i].y / 2, dp[i - 1][max(j - n[i].x, 0)][1] + n[i].y});
		} 
	}
	
	cout << min(dp[a][b][1], dp[a][b][0]) << endl;	
	
	return 0;
}
```
