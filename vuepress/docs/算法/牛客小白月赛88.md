## 超级闪光牛可乐

> 思路：只用根据最大值来判断即可

```c++
#include<bits/stdc++.h> 
using namespace std;

int main(){
	int a, n;
	cin >> a >> n;
	if(n == 0){
		cout << 0 << endl;
		return 0;
	}
	int max_ = 0;
	char maxc;
	for(int i = 0; i < n; i++){
		char c;
		int x;
		cin >> c >> x;
		if(x > max_){
			max_ = x;
			maxc = c;
		}
	}
	int cc = ceil(a * 1.0 / max_);
	if(cc > 1000) cout << -1 << endl;
	else if (cc == 0){
		cout << maxc << endl;
	}
	else {
		string str = "";
		for(int i = 0; i < cc; i++) str += maxc;
		cout << str << endl;
	}
	 
	return 0;
}
```

## 人列计算机

> 简单模拟

```c++
#include<bits/stdc++.h> 
using namespace std;

int main(){
	char n[6][11];
	for(int i = 0; i < 5; i++){
		string str;
		getline(cin, str);
		for(int j = 0; j < str.size(); j++){
			n[i][j] = str[j];
		}
	}
	if(n[2][5] == '&'){
		char a = n[1][0], b = n[3][0];
		cout << ((a - '0') & (b - '0')) << endl;
	} else if(n[2][5] == '='){
		char a = n[1][0], b = n[3][0];
		cout << ((a - '0') | (b - '0')) << endl;
	} else {
		char a = n[2][0];
		cout << ((a - '0') == 1 ? 0 : 1) << endl;
	}
	 
	return 0;
}
```

## 我不是大富翁

> 好家伙，从这就开始上强度了。一开始使用bfs和set模拟两种方法都超时；看题解需要使用二维dp。
>
> 思路：不断刷新数组v的状态

```c++
#include<bits/stdc++.h> 
using namespace std;
int a, b;

int main(){
	cin >> a >> b;
	vector<int> v(a), n[5001];
	for(int i = 0; i < b; i++){
		cin >> n[i]; 
	}
	v[0] = 1;
	for(int i = 0; i < b; i++){
		vector<int> tmp(a);
		n[i] %= a;
		for(int j = 0; j < a; j++){
			if(v[j] == 1){
				tmp[(j + n[i]) % a] = 1;
				tmp[(j - n[i] + a) % a] = 1;
			}
		}	
		v.swap(tmp);
	}
	if(v[0] == 1){
		cout << "YES" << endl;
	} else{
		cout << "NO" << endl;
	}
	 
	return 0;
}
```
