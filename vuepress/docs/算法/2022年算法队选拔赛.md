## A - Numbers

签到题，很清除的记得当时没看清楚题要求，输出化简后的格式，一直没化简....

 ```c++
 #include<bits/stdc++.h>
 #define endl "\n"

 typedef long long ll;
 typedef pair<int, int> PII;
 const int N = 1e6 + 10, M = 1e9 + 10;
 int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
 int a, b, x, y, z, res;
 
 int fun(int x, int y){
 	return y ? fun(y, x % y) : x;
 }
 
 int main(){
 	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
 	cin >> a;
 	for(int i = 2; i <= a - 1; i ++){
 		int num = a;
 		while(num){
 			res += num % i;
 			num /= i;
 		} 
 	}
 	int cc = fun(res, a - 2);
 	cout << res / cc << "/" << (a - 2) / cc << endl; 
 	
 	return 0;
 }
 ```

## B - 表达式求值

不用考虑括号写起来就很nice

 ```c++
 #include<bits/stdc++.h>
 #define endl "\n"
 using namespace std;
 typedef long long ll;
 typedef pair<int, int> PII;
 const int N = 1e6 + 10, M = 1e9 + 10;
 int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
 int a, b, x, y, z;
 ll res;
 stack<ll> s1;
 stack<char> s2;
 unordered_map<char, int> ump;
 
 void init(){
 	ump['+'] = 1;
 	ump['*'] = 2;
 }
 
 void fun(char c){
 	while(s2.size() && s1.size() >= 2 && ump[s2.top()] >= ump[c]){
 		ll num1 = s1.top(), cot = 0;
 		s1.pop();
 		ll num2 = s1.top();
 		s1.pop();
 		char c = s2.top();
 		s2.pop();
 		if(c == '+') cot = num1 + num2;
 		else cot = num1 * num2;
 		s1.push(cot);
 	}	
 	s2.push(c);	
 }
 
 void end(){
 	while(s2.size() && s1.size() >= 2){
 		ll num1 = s1.top(), cot = 0;
 		s1.pop();
 		ll num2 = s1.top();
 		s1.pop();
 		char c = s2.top();
 		s2.pop();
 		if(c == '+') cot = num1 + num2;
 		else cot = num1 * num2;
 		s1.push(cot);
 	}	
 }
 
 int main(){
 	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
 	init();
 	string str;
 	cin >> str;
 	for(int i = 0; i < str.size(); i++){
 		int idx = i;
 		ll num = 0;
 		while(str[idx] >= '0' && str[idx] <= '9'){
 			num = num * 10 + str[idx] - '0';
 			idx ++;
 		}
 		if(num != 0){
 			s1.push(num);
 			i = idx - 1;
 			continue;
 		}
 		if(str[i] == '+' || str[i] == '*'){
 			fun(str[i]);
 		}
 	}
 	end();
 	cout << s1.top() % 10000 << endl;
 	
 	return 0;
 }
 ```

## C - T-primes

数据量很大，需要用到质数筛

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, z, res;
int n[N], st[N], idx;

void init(){
	st[1] = 1;
	for(int i = 2; i <= N; i ++){
		if(!st[i]) n[idx ++] = i; 
		for(int j = 0; n[j] <= N / i; j ++){
			st[n[j] * i] = 1;
			if(i % n[j] == 0) break;
		}
	}
}

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	init();
	cin >> a;
	for(int i = 0; i < a; i++){
		ll num;
		cin >> num;
		ll cc = sqrt(num);
		if(cc * cc == num && !st[cc]){
			cout << "YES" << endl; 
		} else {
			cout << "NO" << endl;
		}
	}
    
	return 0;
}
```

## E - 最小新整数

 单调栈

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, z, res;

string get(stack<char> s){
	string res = "";
	while(s.size()){
		res = s.top() + res;
		s.pop();
	} 
	return res;
}

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	string str;
	cin >> a;
	while(a --){
		cin >> str >> x;
		stack<char> s;
		int len = str.size(), cot = 0;
		for(int i = 0; i < len; i++){
			while(!s.empty() && s.top() > str[i] && cot < x){
				cot ++;
				s.pop();
			}
			s.push(str[i]);
		}
		if(cot == x){
			cout << get(s) << endl;
		} else {
			int num = x - cot;
			string str = get(s);
			for(int i = 0; i < str.size() - num; i++){
				cout << str[i];
			}
			cout << endl;
		}
	}
    
	return 0;
}
```
