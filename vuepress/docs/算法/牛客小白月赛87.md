## 小苯的石子游戏

## 小苯的IDE括号问题（easy）

使用两字符数组分别模拟 `I` 前后两个字符串的操作

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, res = 0;
string str, s;
char v[N], m[N]; 
int len1, len2, c1, c2;

void fun1(){
	if(len1 >= 0 && len2 < c2){
		if(v[len1] == '(' && m[len2] == ')'){
			len1 --;
			len2 ++;
		} else {
			len1 --;
		}	
	} else {
		len1 --;
	}
	
}

void fun2(){
	len2 ++;
}

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a >> b;
	cin >> str;
	int idx = 0, len = str.size();
	for(int i = 0; i < len; i++){
		if(str[i] == 'I'){
			idx = i;
			break;
		}
		v[i] = str[i];
		c1 ++;
	}
	for(int i = idx + 1; i < len; i++){
		m[i - idx - 1] = str[i];
		c2 ++;
	}
	len1 = c1 - 1, len2 = 0;
	while(b --){
		cin >> s;
		if(s[0] == 'b'){
			fun1();
		} else {
			fun2();
		}
	} 

	for(int i = 0; i <= len1; i++) cout << v[i];
	cout << "I";
	for(int i = len2; i < c2; i++) cout << m[i]; 
	
	
	return 0;
}
```

## 小苯的IDE括号问题（hard）

这题与前一题的区别在于操作数增加了两个左移和右移，若再次使用数组的话不好模拟（需要遍历），于是采用两个双端队列进模拟（上一题也可以使用双端队列进行模拟）

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, res = 0;
string str, s;
deque<char> d1, d2;

void fun1(){
	int len1 = d1.size(), len2 = d2.size();
	if(len1 && len2){
		if(d1.back() == '(' && d2.front() == ')'){
			d1.pop_back();
			d2.pop_front();
		} else {
			d1.pop_back();
		}
	} else if(len1) {
		d1.pop_back();
	}
	
}

void fun2(){
	if(d2.size()){
		d2.pop_front();
	}
}

void fun3(){
	if(d1.size()){
		char c = d1.back();
		d1.pop_back(); 
		d2.push_front(c);
	}
}

void fun4(){
	if(d2.size()){
		char c = d2.front();
		d2.pop_front();
		d1.push_back(c);
	}
}

int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a >> b;
	cin >> str;
	int idx = 0, len = a;
	for(int i = 0; i < len; i++){
		if(str[i] == 'I'){
			idx = i;
			break;
		}
		d1.push_back(str[i]); 
	}
	for(int i = idx + 1; i < len; i++){
		d2.push_back(str[i]);
	}
	
	while(b --){
		cin >> s;
		if(s[0] == 'b'){
			fun1();
		} else if(s[0] == 'd'){
			fun2();
		} else if(s[0] == '<'){
			fun3();
		} else{
			fun4();
		}
	} 

	while(d1.size()){
		cout << d1.front();
		d1.pop_front();
	}
	cout << "I";
	while(d2.size()){
		cout << d2.front();
		d2.pop_front();
	}
		
	return 0;
}
```

## 小苯的数组构造

遍历每个数，找出每个数前的各个最大值（不断维护），相减即可

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, res;


int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a;
	vector<ll> v(a + 1);
	v[0] = -10e9 - 10;
	int flag = 1; 
	for(int i = 1; i <= a; i++) {
		cin >> v[i];
		if(i != 0 && v[i] < v[i - 1]){
			flag = 0;
		} 	
	}
	if(flag) {
		for(int i = 1; i <= a; i++){
			cout << 1 << " ";
		}
		return 0;
	}
	for(int i = 1; i <= a; i++){
		if(v[i] >= v[i - 1]) cout << 0 << " ";
		else {
			cout << v[i - 1] - v[i] << " ";
			v[i] = v[i - 1];
		}
	}	
			
	return 0;
}
```

## 小苯的数组切分

思维题，分别考虑 & ^ | 这三种运算符

- 对于 &，无论有多少个数相互 & 操作，结果都是越来越小
- 对于 ^，若有两个相同的数进行 ^，则结果为0

只要知道上面这两个，那么就可以确定右端点这一个数进行 &，剩下的 0, a - 1区间就用维护出 ^ 和 | 这两种操作的最大之和，加上右端点的值就是结果

> 对于维护 0, a - 1这个区间，可以考虑先遍历 0 , a - 1，对每个数进行取 ^ 运算得到一个结果l，再反向遍历 a - 1, 0 若想不让这个数取 ^，可以对 l 对这个数进行取 ^ ，因为开始已经取过一边了，再取一边就是0相当于没取。

```c++
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
typedef pair<int, int> PII;
const int N = 1e6 + 10, M = 1e9 + 10;
int ix[4] = {1,-1,0,0}, iy[4] = {0,0,1,-1};
int a, b, x, y, res;


int main(){
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> a;
	ll res = 0;
	vector<int> v(a);
	for(int i = 0; i < a; i++) cin >> v[i]; 
	ll l = v[0], r = v[a - 2];
	for(int i = 1; i < a - 1; i++) l ^= v[i];
	for(int i = a - 2; i >= 0; i--){
		l ^= v[i];
		r |= v[i];
		res = max(res, l + r);
	}
	
	cout << res + v[a - 1] << endl;
	
	return 0;
}
```
